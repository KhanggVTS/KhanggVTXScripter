local function createButton(tabIndex, name, icon, callback)
    local tab = tabs[tabIndex]
    if not tab then return end
    
    local btn = Instance.new("TextButton", tab.page)
    btn.Size = UDim2.new(0, 160, 0, 45)
    btn.BackgroundColor3 = Color3.fromRGB(30, 30, 40)
    btn.BackgroundTransparency = 0.3
    btn.Text = icon .. " " .. name
    btn.TextColor3 = Color3.new(1,1,1)
    btn.Font = Enum.Font.GothamSemibold
    btn.TextScaled = true
    btn.AutoButtonColor = false
    btn.ZIndex = 102
    
    Instance.new("UICorner", btn).CornerRadius = UDim.new(0, 6)
    local stroke = Instance.new("UIStroke", btn)
    stroke.Color = Color3.fromRGB(0, 255, 200)
    stroke.Thickness = 1
    stroke.Transparency = 0.8
    
    btn.MouseEnter:Connect(function()
        TweenService:Create(btn, TweenInfo.new(0.15), {
            BackgroundColor3 = Color3.fromRGB(50, 50, 70)
        }):Play()
    end)
    btn.MouseLeave:Connect(function()
        TweenService:Create(btn, TweenInfo.new(0.15), {
            BackgroundColor3 = Color3.fromRGB(30, 30, 40)
        }):Play()
    end)
    btn.MouseButton1Click:Connect(function()
        TweenService:Create(btn, TweenInfo.new(0.1), {
            BackgroundColor3 = Color3.fromRGB(0, 255, 200),
            TextColor3 = Color3.new(0,0,0)
        }):Play()
        task.wait(0.1)
        TweenService:Create(btn, TweenInfo.new(0.2), {
            BackgroundColor3 = Color3.fromRGB(50, 50, 70),
            TextColor3 = Color3.new(1,1,1)
        }):Play()
        safeCall(callback)
    end)
    
    tab.page.CanvasSize = UDim2.new(0, 0, 0, tab.grid.AbsoluteContentSize.Y + 20)
    tab.grid:GetPropertyChangedSignal("AbsoluteContentSize"):Connect(function()
        tab.page.CanvasSize = UDim2.new(0, 0, 0, tab.grid.AbsoluteContentSize.Y + 20)
    end)
    
    return btn
end

--// DRAG FUNCTIONALITY
local dragging, dragStart, startPos

titleBar.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 then
        dragging = true
        dragStart = input.Position
        startPos = main.Position
    end
end)

UIS.InputEnded:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 then
        dragging = false
    end
end)

UIS.InputChanged:Connect(function(input)
    if dragging and input.UserInputType == Enum.UserInputType.MouseMovement then
        local delta = input.Position - dragStart
        main.Position = UDim2.new(
            startPos.X.Scale,
            startPos.X.Offset + delta.X,
            startPos.Y.Scale,
            startPos.Y.Offset + delta.Y
        )
    end
end)

    elseif input.KeyCode == Enum.KeyCode.X then
        states.noclip = not states.noclip
        notify("NOCLIP", states.noclip and "ON" or "OFF", 0.5)
    elseif input.KeyCode == Enum.KeyCode.C then
        states.clickTP = not states.clickTP
        notify("CLICK TP", states.clickTP and "ON" or "OFF", 0.5)
    elseif input.KeyCode == Enum.KeyCode.G then
        states.godMode = not states.godMode
        notify("GOD MODE", states.godMode and "ON" or "OFF", 0.5)
    elseif input.KeyCode == Enum.KeyCode.V then
        states.invisible = not states.invisible
        local c = getChar()
        for _, p in pairs(c:GetDescendants()) do
            if p:IsA("BasePart") then
                p.Transparency = states.invisible and 1 or 0
            end
        end
        notify("INVISIBLE", states.invisible and "ON" or "OFF", 0.5)
    end
end)
