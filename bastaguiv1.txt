--// ANTI-CRASH PROTECTION
local function safeCall(func, ...)
    local success, result = pcall(func, ...)
    if not success then warn("BASTA ERROR: ", result) end
    return success, result
end

--// GET CHAR SAFE (CACHE)
local currentChar, currentHum, currentHRP
local function getChar()
    if not player.Character or not player.Character:FindFirstChild("Humanoid") or not player.Character:FindFirstChild("HumanoidRootPart") then
        currentChar = player.Character or player.CharacterAdded:Wait()
        currentHum = currentChar:WaitForChild("Humanoid")
        currentHRP = currentChar:WaitForChild("HumanoidRootPart")
    else
        currentChar = player.Character
        currentHum = currentChar.Humanoid
        currentHRP = currentChar.HumanoidRootPart
    end
    return currentChar, currentHum, currentHRP
end

player.CharacterAdded:Connect(function(char)
    currentChar = char
    currentHum = char:WaitForChild("Humanoid")
    currentHRP = char:WaitForChild("HumanoidRootPart")
end)

    end

    if input.UserInputType == Enum.UserInputType.MouseButton1 then
        draggingToggle = false
    end

    if draggingToggle and input.UserInputType == Enum.UserInputType.MouseMovement then
        local delta = input.Position - toggleDragStart
        toggleBtn.Position = UDim2.new(
            toggleStartPos.X.Scale,
            toggleStartPos.X.Offset + delta.X,

end)

        guiHidden = not guiHidden
        if main then

            toggleBtn.Text = guiHidden and "👁️" or "😈"
        end

            notify("FLY", "Disabled (Keybind)", 1)
        else
            states.fly = true
            notify("FLY", "Enabled (Keybind)", 1)

        end

        if states.noclip then
            states.noclip = false
        else
            states.noclip = true
        end
        notify("NOCLIP", states.noclip and "Enabled (Keybind)" or "Disabled", 1)
    end
end)
