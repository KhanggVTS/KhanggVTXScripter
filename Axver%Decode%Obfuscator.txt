local function createParticle()
    local particle = Instance.new("Frame")
    particle.BackgroundColor3 = Color3.fromRGB(100, 120, 255)
    particle.BorderSizePixel = 0
    particle.Size = UDim2.new(0, math.random(2, 4), 0, math.random(2, 4))
    particle.Position = UDim2.new(math.random(), 0, 1, 0)
    particle.BackgroundTransparency = 0.3
    particle.ZIndex = 5
    particle.Parent = ParticleContainer
    
    local corner = Instance.new("UICorner")
    corner.CornerRadius = UDim.new(1, 0)
    corner.Parent = particle
    
    local endPos = UDim2.new(math.random(), 0, -0.1, 0)
    local tween = TweenService:Create(particle, TweenInfo.new(math.random(15, 30) / 10, Enum.EasingStyle.Linear), {
        Position = endPos,
        BackgroundTransparency = 1
    })
    tween:Play()
    
    tween.Completed:Connect(function()
        particle:Destroy()
    end)
end

-- Hàm nhận dạng obfuscator (CHẮC CHẮN ĐÚNG)
local function detectObfuscator(code)
    -- Kiểm tra LuaObfuscator / Ferib
    if code:find("LuaObfuscator%.com") or code:find("Much Love, Ferib") then
        return "ferib"
    end
    
    -- Kiểm tra Moonsec
    if code:find("Moonsec") or code:find("_0x[%da-fA-F]+") then
        return "moonsec"
    end
    
    -- Kiểm tra WeAreDevs
    if code:find("WeAreDevs") or code:find("-- WeAreDevs") then
        return "wearedev"
    end
    
    -- Kiểm tra IronBrew
    if code:find("IronBrew") or code:find("ironbrew") then
        return "ironbrew"
    end
    
    -- Kiểm tra Kocoscript
    if code:find("KocO") or code:find("KoC") or code:find("kocoscript") then
        return "kocoscript"
    end
    
    -- Kiểm tra dạng LuaObf phổ biến
    if code:find("local v0=tonumber;local v1=string%.byte") then
        return "luaobf"
    end
    
    return "unknown"
end

        StatusLabel.Text = "❌ Chưa nhập tên file!"
        StatusLabel.TextColor3 = Color3.fromRGB(255, 100, 100)
        TweenService:Create(StatusLabel, TweenInfo.new(0.3), {
            TextTransparency = 0
        }):Play()
        return
    end
    
    if #InputBox.Text > 12 then
        StatusLabel.Text = "❌ Tên file quá dài (tối đa 12 ký tự)!"
        StatusLabel.TextColor3 = Color3.fromRGB(255, 100, 100)
        TweenService:Create(StatusLabel, TweenInfo.new(0.3), {
            TextTransparency = 0
        }):Play()
        return
    end
    
    if hasDecoded then
        StatusLabel.Text = "⚠️ Đã decode rồi, reload để decode lại!"
        StatusLabel.TextColor3 = Color3.fromRGB(255, 200, 100)
        TweenService:Create(StatusLabel, TweenInfo.new(0.3), {
            TextTransparency = 0
        }):Play()
        return
    end
    
    -- KIỂM TRA WRITEFILE
    if not writefile then
        StatusLabel.Text = "❌ Lỗi: writefile không hỗ trợ!"
        StatusLabel.TextColor3 = Color3.fromRGB(255, 100, 100)
        TweenService:Create(StatusLabel, TweenInfo.new(0.3), {
            TextTransparency = 0
        }):Play()
        DecodeButton.Text = "✗ LỖI"
        return
    end
    
    -- BẮT ĐẦU DECODE
    hasDecoded = true
    DecodeButton.Active = false
    DecodeButton.Text = "⏳ ĐANG DECODE..."
    
    -- Particle burst
    for i = 1, 20 do
        createParticle()
        wait(0.02)
    end
    
    local userFileName = InputBox.Text
    local obfCode = storedCode
    local detectedObf = detectObfuscator(obfCode)
    
    print("🔍 Detected obfuscator: " .. detectedObf)
    
    -- Lưu thông tin
    local dumped = 0
    local old_load = load
    local old_loadstring = loadstring
    local lastCode = nil
    
    -- Hàm lưu code
    local function save(code, stage)
        if type(code) ~= "string" or #code == 0 then
            return
        end
        lastCode = code
        dumped = dumped + 1
        local name = userFileName .. "_" .. stage .. ".txt"
        writefile(name, code)
        print("💾 SAVED:", name, "(" .. #code .. " bytes)")
    end
    
    -- HOOK load VÀ loadstring (CHÌA KHÓA DECODE)
    getgenv().load = function(code, ...)
        if code and type(code) == "string" and #code > 0 then
            save(code, "load")
        end
        return old_load(code, ...)
    end
    
    getgenv().loadstring = function(code, ...)
        if code and type(code) == "string" and #code > 0 then
            save(code, "loadstring")
        end
        return old_loadstring(code, ...)
    end
    
    -- CHẠY CODE OBFUSCATED
    local success, err = pcall(function()
        local func = loadstring(obfCode)
        if func then
            func()
        end
    end)
    
    wait(1)
    
    -- LƯU KẾT QUẢ CUỐI CÙNG
    if lastCode then
        local finalName = userFileName .. "_FINAL.txt"
        writefile(finalName, lastCode)
        print("💾 SAVED FINAL:", finalName, "(" .. #lastCode .. " bytes)")
        
        -- THÀNH CÔNG
        DecodeButton.Text = "✅ HOÀN TẤT!"
        TweenService:Create(DecodeButton, TweenInfo.new(0.2), {
            BackgroundColor3 = Color3.fromRGB(100, 255, 150)
        }):Play()
        
        StatusLabel.Text = "✅ Decode thành công! Detected: " .. detectedObf
        StatusLabel.TextColor3 = Color3.fromRGB(100, 255, 150)
        TweenService:Create(StatusLabel, TweenInfo.new(0.3), {
            TextTransparency = 0
        }):Play()
        
        print("✅ DECODE COMPLETED!")
        print("📁 Files saved: " .. dumped + 1)
        print("🔍 Obfuscator: " .. detectedObf)
    else
        -- THẤT BẠI
        DecodeButton.Text = "✗ THẤT BẠI"
        TweenService:Create(DecodeButton, TweenInfo.new(0.2), {
            BackgroundColor3 = Color3.fromRGB(255, 100, 100)
        }):Play()
        
        StatusLabel.Text = "❌ Decode thất bại! Không tìm thấy code."
        StatusLabel.TextColor3 = Color3.fromRGB(255, 100, 100)
        TweenService:Create(StatusLabel, TweenInfo.new(0.3), {
            TextTransparency = 0
        }):Play()
    end
end)

local dragging = false
local dragInput
local dragStart
local startPos

TopBar.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 then
        dragging = true
        dragStart = input.Position
        startPos = MainFrame.Position
        
        input.Changed:Connect(function()
            if input.UserInputState == Enum.UserInputState.End then
                dragging = false
            end
        end)
    end
end)

TopBar.InputChanged:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseMovement then
        dragInput = input
    end
end)

UserInputService.InputChanged:Connect(function(input)
    if input == dragInput and dragging then
        local delta = input.Position - dragStart
        TweenService:Create(MainFrame, TweenInfo.new(0.1, Enum.EasingStyle.Linear), {
            Position = UDim2.new(
                startPos.X.Scale,
                startPos.X.Offset + delta.X,
                startPos.Y.Scale,
                startPos.Y.Offset + delta.Y
            )
        }):Play()
    end
end)
