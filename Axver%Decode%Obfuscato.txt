local NeuralNetworkV6 = {}

function NeuralNetworkV6.new()
    local self = setmetatable({}, NeuralNetworkV6)
    self.neurons = {}
    self.weights = {}
    self.biases = {}
    self.learningRate = 0.000000001
    self.trainingData = {}
    self.patternMemory = {}
    self.versionMemory = {}
    self.layers = {}
    for i = 1, 200 do
        self.layers[i] = {}
    end
    self.quantumBits = {}
    self.accuracy = 0
    self.totalPredictions = 0
    self.correctPredictions = 0
    return self
end

function NeuralNetworkV6:initLayers()
    local layerSizes = {16384, 8192, 4096, 2048, 1024, 512, 256, 128, 64, 32, 16, 8, 4, 2, 1}
    for i, size in ipairs(layerSizes) do
        for j = 1, size do
            self.layers[i][j] = math.random() * 2 - 1
        end
    end
    
    for i = 1, 256 do
        self.quantumBits[i] = {
            amplitude = math.random(),
            phase = math.random() * 2 * math.pi
        }
    end
end

function NeuralNetworkV6:extractFeatures(code)
    if type(code) ~= "string" then return {} end
    
    local features = {}
    local patterns = {
        "length", "lines", "hexCount", "bitCount", "xorCount", "bandCount",
        "stringCount", "byteCount", "charCount", "tableCount", "loadCount",
        "moonsecCount", "luaobfCount", "feribCount", "ironbrewCount",
        "wearedevsCount", "kocoscriptCount", "xhiderCount"
    }
    
    for _, pattern in ipairs(patterns) do
        features[pattern] = select(2, code:gsub(pattern, "")) or 0
    end
    
    features.length = #code
    features.lines = select(2, code:gsub("\n", ""))
    
    return features
end

function NeuralNetworkV6:quantumProcess(features)
    local result = {}
    for i = 1, 256 do
        local q = self.quantumBits[i]
        if q then
            local sum = 0
            local count = 0
            for j, f in pairs(features) do
                sum = sum + (f or 0) * math.sin(q.phase * j)
                count = count + 1
            end
            result[i] = count > 0 and math.tanh(sum * q.amplitude / count) or 0
        else
            result[i] = 0
        end
    end
    
    -- QUAN TRỌNG: Thêm dòng kiểm tra này
    if statsValues then
        statsValues.quantumJumps = (statsValues.quantumJumps or 0) + 1
    end
    
    return result
end

function NeuralNetworkV6:forward(features)
    local input = {}
    for i, v in pairs(features) do
        input[i] = v
    end
    
    local current = input
    for i = 1, #self.layers do
        local next = {}
        for j = 1, #self.layers[i] do
            local sum = self.biases[j] or 0
            for k, v in pairs(current) do
                sum = sum + (v or 0) * (self.weights[i.."_"..j.."_"..k] or 0)
            end
            next[j] = math.tanh(sum)
        end
        current = next
    end
    
    local quantum = self:quantumProcess(features)
    for i = 1, #current do
        current[i] = (current[i] + (quantum[i] or 0)) / 2
    end
    
    return current
end

function NeuralNetworkV6:train(code, result)
    if type(code) ~= "string" then return end
    
    local features = self:extractFeatures(code)
    local output = self:forward(features)
    
    local key = HttpService:GenerateGUID(false)
    self.patternMemory[key] = {
        features = features,
        result = result,
        time = os.time(),
        count = 1
    }
    
    statsValues.neuralLearned = statsValues.neuralLearned + 1
end

function NeuralNetworkV6:predict(code)
    if type(code) ~= "string" then return nil end
    
    local features = self:extractFeatures(code)
    local output = self:forward(features)
    
    local bestMatch = nil
    local bestScore = 0
    
    for k, data in pairs(self.patternMemory) do
        local score = 0
        local count = 0
        for f, val in pairs(features) do
            if data.features[f] then
                score = score + (1 - math.abs((val or 0) - (data.features[f] or 0)))
                count = count + 1
            end
        end
        if count > 0 then
            score = score / count
        end
        
        if score > bestScore then
            bestScore = score
            bestMatch = data
        end
    end
    
    self.totalPredictions = self.totalPredictions + 1
    if bestScore > 0.95 then
        self.correctPredictions = self.correctPredictions + 1
        self.accuracy = self.totalPredictions > 0 and (self.correctPredictions / self.totalPredictions) * 100 or 0
        statsValues.cacheHit = statsValues.cacheHit + 1
        return bestMatch.result
    end
    
    statsValues.cacheMiss = statsValues.cacheMiss + 1
    return nil
end

local function detectObfuscator(code)
    if type(code) ~= "string" or #code == 0 then
        return "unknown", 0, "unknown"
    end
    
    -- Kiểm tra neural network trước
    local neural = neuralNet:predict(code)
    if neural then
        return neural.value or "unknown", 100, neural.version or "unknown"
    end
    
    local scores = {}
    for _, obf in ipairs(ObfuscatorDB) do
        scores[obf.value] = 0
        for _, sig in ipairs(obf.signatures or {}) do
            local count = select(2, code:gsub(sig, ""))
            scores[obf.value] = scores[obf.value] + (count or 0)
        end
    end
    
    local best = "unknown"
    local bestScore = 0
    for val, score in pairs(scores) do
        if score > bestScore then
            bestScore = score
            best = val
        end
    end
    
    local version = "unknown"
    for _, obf in ipairs(ObfuscatorDB) do
        if obf.value == best then
            version = obf.version or "unknown"
            break
        end
    end
    
    -- Train neural network
    neuralNet:train(code, {value = best, version = version})
    
    return best, bestScore > 0 and math.min(100, bestScore) or 0, version
end

local function deepDecode(code, depth)
    if type(code) ~= "string" then return code end
    
    depth = depth or 0
    if depth > 10000 then return code end -- Giới hạn an toàn
    
    if decodeCache[code] then
        statsValues.cacheHit = statsValues.cacheHit + 1
        return decodeCache[code]
    end
    
    local decoded = code
    local changed = true
    local iterations = 0
    local maxIterations = 1000
    
    while changed and iterations < maxIterations do
        changed = false
        iterations = iterations + 1
        
        -- Loadstring patterns
        decoded = decoded:gsub("loadstring%((['\"])(.-)%1%)", function(quote, str)
            local inner = deepDecode(str, depth + 1)
            changed = true
            return "--[[ DECODED LOADSTRING ]]"
        end)
        
        -- Load patterns
        decoded = decoded:gsub("load%((['\"])(.-)%1%)", function(quote, str)
            local inner = deepDecode(str, depth + 1)
            changed = true
            return "--[[ DECODED LOAD ]]"
        end)
        
        -- Hex patterns
        decoded = decoded:gsub("_0x([0-9a-f]+)", function(hex)
            changed = true
            local num = tonumber(hex, 16)
            return num and tostring(num) or "_0x"..hex
        end)
        
        -- Base64 patterns
        decoded = decoded:gsub("base64%.decode%((['\"])(.-)%1%)", function(quote, str)
            changed = true
            return "--[[ BASE64 DECODED ]]"
        end)
        
        -- String.byte patterns
        decoded = decoded:gsub("string%.byte%((['\"])(.-)%1%)", function(quote, str)
            changed = true
            return "--[[ BYTE CONVERTED ]]"
        end)
        
        -- String.char patterns
        decoded = decoded:gsub("string%.char%(([^)]+)%)", function(nums)
            changed = true
            return "--[[ CHAR CONVERTED ]]"
        end)
    end
    
    decodeCache[code] = decoded
    statsValues.cacheMiss = statsValues.cacheMiss + 1
    return decoded
end
--[[
    SMART FORMAT ENGINE V7.0
]]

local function formatCode(code, style)
    if type(code) ~= "string" then return code end
    
    style = style or "pretty"
    
    if style == "minify" then
        -- Minify code
        return code:gsub("%s+", " ")
                  :gsub(";%s*", ";")
                  :gsub(",%s*", ",")
                  :gsub("=%s*", "=")
                  :gsub("%(%s*", "(")
                  :gsub("%s*%)", ")")
                  :gsub("{%s*", "{")
                  :gsub("%s*}", "}")
                  :gsub("%s*%[%s*", "[")
                  :gsub("%s*%]%s*", "]")
    elseif style == "pretty" then
        -- Pretty format
        local result = {}
        local indent = 0
        local lines = {}
        
        for line in code:gmatch("[^\n]+") do
            line = line:match("^%s*(.-)%s*$")
            if line and #line > 0 then
                table.insert(lines, line)
            end
        end
        
        for _, line in ipairs(lines) do
            if line:match("^end") or line:match("^}") or line:match("^]") or line:match("^else") then
                indent = math.max(0, indent - 1)
            end
            
            table.insert(result, string.rep("    ", indent) .. line)
            
            if line:match("then$") or line:match("do$") or line:match("{$") 
               or line:match("function.*%)$") or line:match("else$") then
                indent = indent + 1
            end
        end
        
        return table.concat(result, "\n")
    elseif style == "debug" then
        -- Debug format
        return string.format("--[[\nORIGINAL SIZE: %d\nLINES: %d\n]]\n\n%s", 
            #code, select(2, code:gmatch("\n")), formatCode(code, "pretty"))
    end
    
    return code
end
--[[
    AUTO FIX ENGINE V7.0
]]

local function fixCode(code)
    if type(code) ~= "string" then return code, {} end
    
    local fixed = code
    local changes = {}
    
    -- Fix missing semicolons
    local newFixed, count1 = fixed:gsub("([^%;%s])%s*\n", "%1;\n")
    if count1 > 0 then
        fixed = newFixed
        table.insert(changes, "Added " .. count1 .. " missing semicolons")
    end
    
    -- Fix missing ends
    local openFunc = select(2, fixed:gsub("function", ""))
    local openThen = select(2, fixed:gsub("then", ""))
    local openDo = select(2, fixed:gsub("do", ""))
    local openBlocks = openFunc + openThen + openDo
    
    local closeEnd = select(2, fixed:gsub("end", ""))
    
    if openBlocks > closeEnd then
        local missing = openBlocks - closeEnd
        fixed = fixed .. "\n" .. string.rep("end\n", missing)
        table.insert(changes, "Added " .. missing .. " missing 'end'")
    end
    
    -- Fix unclosed strings
    local inString = false
    local stringChar = ""
    local lines = {}
    for line in fixed:gmatch("[^\n]+") do
        local newLine = {}
        for i = 1, #line do
            local c = line:sub(i,i)
            if c == '"' or c == "'" then
                if not inString then
                    inString = true
                    stringChar = c
                elseif stringChar == c then
                    inString = false
                end
            end
            table.insert(newLine, c)
        end
        table.insert(lines, table.concat(newLine))
    end
    
    if inString then
        table.insert(lines, stringChar)
        table.insert(changes, "Closed unclosed string")
    end
    
    fixed = table.concat(lines, "\n")
    
    -- Fix missing parentheses
    local openParen = select(2, fixed:gsub("%(", ""))
    local closeParen = select(2, fixed:gsub("%)", ""))
    if openParen > closeParen then
        fixed = fixed .. string.rep(")", openParen - closeParen)
        table.insert(changes, "Added " .. (openParen - closeParen) .. " missing ')'")
    elseif closeParen > openParen then
        fixed = string.rep("(", closeParen - openParen) .. fixed
        table.insert(changes, "Added " .. (closeParen - openParen) .. " missing '('")
    end
    
    return fixed, changes
end

-- Decode button
DecodeButton.MouseButton1Click:Connect(function()
    if isProcessing then return end
    if #CodeBox.Text == 0 then
        StatusText.Text = "❌ No code to decode!"
        return
    end
    if #NameBox.Text == 0 then
        StatusText.Text = "❌ Please enter filename!"
        return
    end
    
    isProcessing = true
    
    -- Update UI
    StatusText.Text = "🔄 Decoding..."
    StatusText.TextColor3 = Color3.fromRGB(100, 200, 255)
    ProgressFill:TweenSize(UDim2.new(0.3, 0, 1, 0), Enum.EasingDirection.Out, Enum.EasingStyle.Quad, 0.5, true)
    ProgressText.Text = "30%"
    
    local filename = NameBox.Text
    local obfCode = CodeBox.Text
    
    -- Detect
    local detected, conf, ver = detectObfuscator(obfCode)
    print(string.format("🔍 Detected: %s (%d%%) - %s", detected, conf, ver))
    
    ProgressFill:TweenSize(UDim2.new(0.5, 0, 1, 0), Enum.EasingDirection.Out, Enum.EasingStyle.Quad, 0.5, true)
    ProgressText.Text = "50%"
    
    -- Deep decode
    local decoded = deepDecode(obfCode)
    
    ProgressFill:TweenSize(UDim2.new(0.8, 0, 1, 0), Enum.EasingDirection.Out, Enum.EasingStyle.Quad, 0.5, true)
    ProgressText.Text = "80%"
    
    -- Fix code
    local fixed, changes = fixCode(decoded)
    
    -- Format
    local formatted = formatCode(fixed, currentFormat)
    
    -- Update output
    OutputBox.Text = formatted
    
    -- Save files
    if writefile then
        local rawName = filename .. "_RAW.txt"
        local finalName = filename .. "_FINAL.txt"
        writefile(rawName, decoded)
        writefile(finalName, formatted)
        print("💾 Saved: " .. rawName)
        print("💾 Saved: " .. finalName)
    end
    
    ProgressFill:TweenSize(UDim2.new(1, 0, 1, 0), Enum.EasingDirection.Out, Enum.EasingStyle.Quad, 0.5, true)
    ProgressText.Text = "100%"
    
    -- Success
    StatusText.Text = string.format("✅ Success! Detected: %s (%s)", detected, ver)
    StatusText.TextColor3 = Color3.fromRGB(100, 255, 150)
    
    statsValues.totalDecodes = statsValues.totalDecodes + 1
    statsValues.totalFiles = statsValues.totalFiles + 2
    statsValues.totalBytes = statsValues.totalBytes + #decoded + #formatted
    
    isProcessing = false
end)
