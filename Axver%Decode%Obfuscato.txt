function NeuralNetworkV9:initLayers()
    local layerSizes = {1024, 512, 256, 128, 64, 32, 16, 8, 4, 2, 1}
    for i, size in ipairs(layerSizes) do
        for j = 1, size do
            self.layers[i][j] = math.random() * 2 - 1
        end
    end
    
    for i = 1, 64 do
        self.quantumBits[i] = {
            amplitude = math.random(),
            phase = math.random() * 2 * math.pi,
            spin = math.random() * 2 - 1,
            entanglement = {}
        }
        for j = 1, 10 do
            self.quantumBits[i].entanglement[j] = math.random()
        end
    end
    
    for i = 1, 32 do
        self.holographicMatrix[i] = {}
        for j = 1, 32 do
            self.holographicMatrix[i][j] = math.random()
        end
    end
end

function NeuralNetworkV9:calculateEntropy(code)
    local freq = {}
    for i = 1, #code do
        local c = code:sub(i,i)
        freq[c] = (freq[c] or 0) + 1
    end
    local entropy = 0
    for _, count in pairs(freq) do
        local p = count / #code
        entropy = entropy - p * math.log(p) / math.log(2)
    end
    return entropy
end

function NeuralNetworkV9:extractFeatures(code)
    if type(code) ~= "string" then return {} end
    
    local features = {}
    local patterns = {
        "length", "lines", "hexCount", "bitCount", "xorCount", "bandCount",
        "stringCount", "byteCount", "charCount", "tableCount", "loadCount",
        "moonsecCount", "luaobfCount", "feribCount", "ironbrewCount",
        "wearedevsCount", "kocoscriptCount", "xhiderCount", "synapseCount",
        "scriptwareCount", "krnlCount", "fluxusCount", "electronCount"
    }
    
    for _, pattern in ipairs(patterns) do
        features[pattern] = select(2, code:gsub(pattern, "")) or 0
    end
    
    features.length = #code
    features.lines = select(2, code:gsub("\n", ""))
    features.entropy = self:calculateEntropy(code)
    
    return features
end

function NeuralNetworkV9:quantumProcess(features)
    local result = {}
    for i = 1, 64 do
        local q = self.quantumBits[i]
        if q then
            local sum = 0
            local count = 0
            for j, f in pairs(features) do
                sum = sum + (f or 0) * math.sin(q.phase * j) * q.spin
                count = count + 1
            end
            for k, e in ipairs(q.entanglement) do
                sum = sum + e * (features[(k % (#features or 1)) + 1] or 0)
            end
            result[i] = count > 0 and math.tanh(sum * q.amplitude / count) or 0
        else
            result[i] = 0
        end
    end
    
    if statsValues then
        statsValues.quantumJumps = (statsValues.quantumJumps or 0) + 1
    end
    
    return result
end

function NeuralNetworkV9:forward(features)
    local input = {}
    for i, v in pairs(features) do
        input[i] = v
    end
    
    local current = input
    for i = 1, #self.layers do
        local next = {}
        for j = 1, #self.layers[i] do
            local sum = self.biases[j] or 0
            for k, v in pairs(current) do
                sum = sum + (v or 0) * (self.weights[i.."_"..j.."_"..k] or 0)
            end
            next[j] = math.tanh(sum)
        end
        current = next
    end
    
    local quantum = self:quantumProcess(features)
    
    for i = 1, #current do
        current[i] = (current[i] + (quantum[i] or 0)) / 2
    end
    
    return current
end

function NeuralNetworkV9:train(code, result)
    if type(code) ~= "string" then return end
    
    local features = self:extractFeatures(code)
    local output = self:forward(features)
    
    local key = HttpService:GenerateGUID(false)
    self.patternMemory[key] = {
        features = features,
        result = result,
        time = os.time(),
        count = 1
    }
    
    if statsValues then
        statsValues.neuralLearned = statsValues.neuralLearned + 1
    end
end

function NeuralNetworkV9:predict(code)
    if type(code) ~= "string" then return nil end
    
    local features = self:extractFeatures(code)
    local output = self:forward(features)
    
    local bestMatch = nil
    local bestScore = 0
    
    for k, data in pairs(self.patternMemory) do
        local score = 0
        local count = 0
        for f, val in pairs(features) do
            if data.features[f] then
                score = score + (1 - math.abs((val or 0) - (data.features[f] or 0)))
                count = count + 1
            end
        end
        if count > 0 then
            score = score / count
        end
        
        if score > bestScore then
            bestScore = score
            bestMatch = data
        end
    end
    
    self.totalPredictions = self.totalPredictions + 1
    if bestScore > 0.95 then
        self.correctPredictions = self.correctPredictions + 1
        self.accuracy = self.totalPredictions > 0 and (self.correctPredictions / self.totalPredictions) * 100 or 0
        if statsValues then
            statsValues.cacheHit = statsValues.cacheHit + 1
        end
        return bestMatch.result
    end
    
    if statsValues then
        statsValues.cacheMiss = statsValues.cacheMiss + 1
    end
    return nil
end

        scores[obf.value] = 0
        for _, sig in ipairs(obf.signatures or {}) do
            local count = select(2, code:gsub(sig, ""))
            if count > 0 then
                local points = count * (#sig > 10 and 5 or 3)
                scores[obf.value] = scores[obf.value] + points
                totalScore = totalScore + points
            end
        end
    end
    
    local best = "unknown"
    local bestScore = 0
    for val, score in pairs(scores) do
        if score > bestScore then
            bestScore = score
            best = val
        end
    end
    
    local version = "unknown"
    for _, obf in ipairs(ObfuscatorDB) do
        if obf.value == best then
            version = obf.version or "unknown"
            break

    end
    
    local confidence = totalScore > 0 and math.min(100, math.floor((bestScore / totalScore) * 100)) or 0
    
    neuralNet:train(code, {value = best, version = version, confidence = confidence})
    
    return best, confidence, version

local function deepDecode(code, depth)
    if type(code) ~= "string" then return code end
    
    depth = depth or 0
    if depth > 1000 then return code end
    
    if decodeCache[code] then
        if statsValues then
            statsValues.cacheHit = statsValues.cacheHit + 1
        end
        return decodeCache[code]
    end
    
    local decoded = code
    local changed = true
    local iterations = 0
    local maxIterations = 100
    
    while changed and iterations < maxIterations do
        changed = false
        iterations = iterations + 1
        
        decoded = decoded:gsub("loadstring%((['\"])(.-)%1%)", function(quote, str)
            local inner = deepDecode(str, depth + 1)
            changed = true
            return "--[[ DECODED LOADSTRING ]]"
        end)
        
        decoded = decoded:gsub("load%((['\"])(.-)%1%)", function(quote, str)
            local inner = deepDecode(str, depth + 1)
            changed = true
            return "--[[ DECODED LOAD ]]"
        end)
        
        decoded = decoded:gsub("_0x([0-9a-f]+)", function(hex)
            changed = true
            local num = tonumber(hex, 16)
            return num and tostring(num) or "_0x"..hex
        end)
        
        decoded = decoded:gsub("base64%.decode%((['\"])(.-)%1%)", function(quote, str)
            changed = true
            return "--[[ BASE64 DECODED ]]"
        end)
        
        decoded = decoded:gsub("string%.byte%((['\"])(.-)%1%)", function(quote, str)
            changed = true
            return "--[[ BYTE CONVERTED ]]"
        end)
    end
    
    decodeCache[code] = decoded
    if statsValues then
        statsValues.cacheMiss = statsValues.cacheMiss + 1
    end
    return decoded
end

local function formatCode(code, style)
    if type(code) ~= "string" then return code end
    
    style = style or "pretty"
    
    if style == "minify" then
        return code:gsub("%s+", " ")
                  :gsub(";%s*", ";")
                  :gsub(",%s*", ",")
                  :gsub("=%s*", "=")
                  :gsub("%(%s*", "(")
                  :gsub("%s*%)", ")")
                  :gsub("{%s*", "{")
                  :gsub("%s*}", "}")
    elseif style == "pretty" then
        local result = {}
        local indent = 0
        local lines = {}
        
        for line in code:gmatch("[^\n]+") do
            line = line:match("^%s*(.-)%s*$")
            if line and #line > 0 then
                table.insert(lines, line)
            end
        end
        
        for _, line in ipairs(lines) do
            if line:match("^end") or line:match("^}") or line:match("^]") or line:match("^else") then
                indent = math.max(0, indent - 1)
            end
            
            table.insert(result, string.rep("    ", indent) .. line)
            
            if line:match("then$") or line:match("do$") or line:match("{$") 
               or line:match("function.*%)$") or line:match("else$") then
                indent = indent + 1
            end
        end
        
        return table.concat(result, "\n")
    elseif style == "debug" then
        return string.format("--[[\nSIZE: %d\nLINES: %d\n]]\n\n%s", 
            #code, 
            select(2, code:gmatch("\n")), 
            formatCode(code, "pretty"))
    end
    
    return code
end
--[[
    AUTO FIX ENGINE V10.0
]]

local function fixCode(code)
    if type(code) ~= "string" then return code, {} end
    
    local fixed = code
    local changes = {}
    
    local newFixed, count1 = fixed:gsub("([^%;%s])%s*\n", "%1;\n")
    if count1 > 0 then
        fixed = newFixed
        table.insert(changes, "Added " .. count1 .. " semicolons")
    end
    
    local openFunc = select(2, fixed:gsub("function", ""))
    local openThen = select(2, fixed:gsub("then", ""))
    local openDo = select(2, fixed:gsub("do", ""))
    local openBlocks = openFunc + openThen + openDo
    
    local closeEnd = select(2, fixed:gsub("end", ""))
    
    if openBlocks > closeEnd then
        local missing = openBlocks - closeEnd
        fixed = fixed .. "\n" .. string.rep("end\n", missing)
        table.insert(changes, "Added " .. missing .. " end")
    end
    
    local inString = false
    local stringChar = ""
    local lines = {}
    for line in fixed:gmatch("[^\n]+") do
        local newLine = {}
        for i = 1, #line do
            local c = line:sub(i,i)
            if c == '"' or c == "'" then
                if not inString then
                    inString = true
                    stringChar = c
                elseif stringChar == c then
                    inString = false
                end
            end
            table.insert(newLine, c)
        end
        table.insert(lines, table.concat(newLine))
    end
    
    if inString then
        table.insert(lines, stringChar)
        table.insert(changes, "Closed string")
    end
    
    fixed = table.concat(lines, "\n")
    
    local openParen = select(2, fixed:gsub("%(", ""))
    local closeParen = select(2, fixed:gsub("%)", ""))
    if openParen > closeParen then
        fixed = fixed .. string.rep(")", openParen - closeParen)
        table.insert(changes, "Added " .. (openParen - closeParen) .. " )")
    end
    
    return fixed, changes
end

DecodeButton.MouseButton1Click:Connect(function()
    if isProcessing then return end
    if #CodeBox.Text == 0 then
        StatusText.Text = "❌ No code to decode!"
        return
    end
    if #NameBox.Text == 0 then
        StatusText.Text = "❌ Please enter filename!"
        return
    end
    
    isProcessing = true
    
    StatusText.Text = "🔄 Decoding..."
    StatusText.TextColor3 = Color3.fromRGB(100, 200, 255)
    ProgressFill:TweenSize(UDim2.new(0.25, 0, 1, 0), Enum.EasingDirection.Out, Enum.EasingStyle.Quad, 0.5, true)
    ProgressText.Text = "25%"
    
    local filename = NameBox.Text
    local obfCode = CodeBox.Text
    
    local detected, conf, ver = detectObfuscator(obfCode)
    print(string.format("🔍 Detected: %s (%d%%)", detected, conf))
    
    ProgressFill:TweenSize(UDim2.new(0.5, 0, 1, 0), Enum.EasingDirection.Out, Enum.EasingStyle.Quad, 0.5, true)
    ProgressText.Text = "50%"
    
    local old_load = load
    local old_loadstring = loadstring
    local old_require = require
    local lastCode = nil
    local dumped = 0
    
    local function saveCode(code, stage)
        if type(code) ~= "string" or #code == 0 then return end
        if decodeCache[code] then return end
        
        decodeCache[code] = true
        lastCode = code
        dumped = dumped + 1
        
        if settings.saveIntermediate then
            local name = string.format("%s_%s_%d.txt", filename, stage, dumped)
            pcall(function() writefile(name, code) end)
        end
    end
    
    getgenv().load = function(code, ...)
        saveCode(code, "load")
        return old_load(code, ...)
    end
    
    getgenv().loadstring = function(code, ...)
        saveCode(code, "loadstring")
        return old_loadstring(code, ...)
    end
    
    getgenv().require = function(module)
        if type(module) == "string" and #module > 100 then
            saveCode(module, "require")
        end
        return old_require(module)
    end
    
    ProgressFill:TweenSize(UDim2.new(0.75, 0, 1, 0), Enum.EasingDirection.Out, Enum.EasingStyle.Quad, 0.5, true)
    ProgressText.Text = "75%"
    
    if settings.deepDecode then
        local deepDecoded = deepDecode(obfCode)
        if deepDecoded ~= obfCode then
            saveCode(deepDecoded, "deep")
        end
    end
    
    local success, err = pcall(function()
        local func = loadstring(obfCode)
        if func then
            local s, e = pcall(func)
            if not s then
                warn("Runtime error:", e)
            end
        end
    end)
    
    wait(0.5)
    
    if lastCode then
        if settings.deepDecode then
            local finalDeep = deepDecode(lastCode)
            if finalDeep ~= lastCode then
                lastCode = finalDeep
            end
        end
        
        local formattedCode = formatCode(lastCode, currentFormat)
        
        local rawName = filename .. "_RAW.txt"
        local finalName = filename .. "_FINAL.txt"
        
        pcall(function() writefile(rawName, lastCode) end)
        pcall(function() writefile(finalName, formattedCode) end)
        
        OutputBox.Text = formattedCode
        OutputScroll.CanvasSize = UDim2.new(0, 0, 0, math.max(400, #formattedCode))
        
        ProgressFill:TweenSize(UDim2.new(1, 0, 1, 0), Enum.EasingDirection.Out, Enum.EasingStyle.Quad, 0.5, true)
        ProgressText.Text = "100%"
        
        StatusText.Text = string.format("✅ Success! Detected: %s", detected)
        StatusText.TextColor3 = Color3.fromRGB(100, 255, 150)
        
        if statsValues then
            statsValues.totalDecodes = statsValues.totalDecodes + 1
            statsValues.totalFiles = statsValues.totalFiles + dumped + 2
            statsValues.totalBytes = statsValues.totalBytes + #lastCode
        end
    else
        StatusText.Text = "❌ Decode failed!"
        StatusText.TextColor3 = Color3.fromRGB(255, 100, 100)
    end
    
    isProcessing = false
end)
