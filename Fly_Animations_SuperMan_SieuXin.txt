local function smoothCameraZoom(targetFOV, targetBlur)
    if zoomTween then
        zoomTween:Cancel()
    end
    
    zoomTween = TweenService:Create(camera, 
        TweenInfo.new(0.6, Enum.EasingStyle.Quad, Enum.EasingDirection.Out),
        {FieldOfView = targetFOV}
    )
    zoomTween:Play()
    
    TweenService:Create(blurEffect, 
        TweenInfo.new(0.6, Enum.EasingStyle.Quad, Enum.EasingDirection.Out),
        {Size = targetBlur}
    ):Play()
end

local function createSpeedLines()
    if speedLines then
        speedLines:Destroy()
        speedLines = nil
    end
    
    speedLines = Instance.new("ScreenGui", pGui)
    speedLines.Name = "SpeedLines"
    speedLines.IgnoreGuiInset = true
    speedLines.ResetOnSpawn = false
    speedLines.ZIndexBehavior = Enum.ZIndexBehavior.Sibling
    
    local frame = Instance.new("Frame", speedLines)
    frame.Size = UDim2.new(1, 0, 1, 0)
    frame.BackgroundTransparency = 1
    frame.ZIndex = 10
    
    for i = 1, 15 do
        local line = Instance.new("Frame", frame)
        line.BackgroundColor3 = Color3.fromRGB(255, 255, 255)
        line.BackgroundTransparency = 0.7
        line.BorderSizePixel = 0
        line.Size = UDim2.new(0, math.random(50, 150), 0, math.random(1, 3))
        line.Position = UDim2.new(math.random(), 0, math.random(), 0)
        line.Rotation = math.random(-20, 20)
        line.ZIndex = 10
        
        task.spawn(function()
            while line and line.Parent and flying do
                TweenService:Create(line, TweenInfo.new(0.3, Enum.EasingStyle.Linear), 
                    {Position = UDim2.new(line.Position.X.Scale - 0.3, 0, line.Position.Y.Scale, 0)}
                ):Play()
                task.wait(0.3)
                if line.Position.X.Scale < -0.2 then
                    line.Position = UDim2.new(1.2, 0, math.random(), 0)
                end
            end
        end)
    end
end

local function setFlyPose(char, speed)
    local hum = char:FindFirstChildOfClass("Humanoid")
    if not hum then return end
    
    if hum.RigType == Enum.HumanoidRigType.R15 then
        local rS = char:FindFirstChild("RightUpperArm") and char.RightUpperArm:FindFirstChild("RightShoulder")
        local lS = char:FindFirstChild("LeftUpperArm") and char.LeftUpperArm:FindFirstChild("LeftShoulder")
        local neck = char:FindFirstChild("Head") and char.Head:FindFirstChild("Neck")
        local rH = char:FindFirstChild("RightUpperLeg") and char.RightUpperLeg:FindFirstChild("RightHip")
        local lH = char:FindFirstChild("LeftUpperLeg") and char.LeftUpperLeg:FindFirstChild("LeftHip")
        local rL = char:FindFirstChild("RightLowerLeg") and char.RightLowerLeg:FindFirstChild("RightKnee")
        local lL = char:FindFirstChild("LeftLowerLeg") and char.LeftLowerLeg:FindFirstChild("LeftKnee")
        local torso = char:FindFirstChild("UpperTorso") and char.UpperTorso:FindFirstChild("Waist")
        
        if rS and lS and neck then
            local armAngle = math.rad(-160 + math.sin(tick() * 8) * 5)
            local legSwing = math.sin(tick() * 6) * 15
            
            if speed > 10 then
                rS.C1 = CFrame.new(0.2, 0.7, -0.2) * CFrame.Angles(armAngle, math.rad(10), math.rad(-25))
                lS.C1 = CFrame.new(-0.2, 0.7, -0.2) * CFrame.Angles(armAngle, math.rad(-10), math.rad(25))
                neck.C1 = CFrame.new(0, -0.3, 0.1) * CFrame.Angles(math.rad(-35), 0, 0)
                
                if rH and lH then
                    rH.C1 = CFrame.new(0, -0.4, 0) * CFrame.Angles(math.rad(-15 + legSwing), 0, math.rad(-5))
                    lH.C1 = CFrame.new(0, -0.4, 0) * CFrame.Angles(math.rad(-15 - legSwing), 0, math.rad(5))
                end
                
                if rL then rL.C1 = CFrame.new(0, -0.5, 0) * CFrame.Angles(math.rad(-10), 0, 0) end
                if lL then lL.C1 = CFrame.new(0, -0.5, 0) * CFrame.Angles(math.rad(-10), 0, 0) end
                if torso then torso.C1 = CFrame.new(0, 0, 0) * CFrame.Angles(math.rad(-25), 0, 0) end
            else
                rS.C1 = rS.C1:Lerp(CFrame.new(0, 0.6, 0) * CFrame.Angles(math.rad(-10), 0, math.rad(-15)), 0.1)
                lS.C1 = lS.C1:Lerp(CFrame.new(0, 0.6, 0) * CFrame.Angles(math.rad(-10), 0, math.rad(15)), 0.1)
                neck.C1 = neck.C1:Lerp(CFrame.new(0, -0.2, 0) * CFrame.Angles(math.rad(-15), 0, 0), 0.1)
                
                if rH then rH.C1 = rH.C1:Lerp(CFrame.new(0, -0.3, 0) * CFrame.Angles(math.rad(-5), 0, 0), 0.1) end
                if lH then lH.C1 = lH.C1:Lerp(CFrame.new(0, -0.3, 0) * CFrame.Angles(math.rad(-5), 0, 0), 0.1) end
            end
        end
    else
        local rS = char:FindFirstChild("Right Shoulder")
        local lS = char:FindFirstChild("Left Shoulder")
        local neck = char:FindFirstChild("Head") and char.Head:FindFirstChild("Neck")
        local rH = char:FindFirstChild("Right Hip")
        local lH = char:FindFirstChild("Left Hip")
        
        if rS and lS and neck then
            local armAngle = math.rad(-150 + math.sin(tick() * 8) * 5)
            local legSwing = math.sin(tick() * 6) * 10
            
            if speed > 10 then
                rS.C1 = CFrame.new(0, 0.5, 0) * CFrame.Angles(armAngle, 0, math.rad(-20))
                lS.C1 = CFrame.new(0, 0.5, 0) * CFrame.Angles(armAngle, 0, math.rad(20))
                neck.C1 = CFrame.new(0, -0.3, 0) * CFrame.Angles(math.rad(-35), 0, 0)
                
                if rH then rH.C1 = CFrame.new(0, -0.5, 0) * CFrame.Angles(math.rad(-10 + legSwing), 0, 0) end
                if lH then lH.C1 = CFrame.new(0, -0.5, 0) * CFrame.Angles(math.rad(-10 - legSwing), 0, 0) end
            else
                rS.C1 = rS.C1:Lerp(CFrame.new(0, 0.5, 0) * CFrame.Angles(math.rad(-10), 0, math.rad(-15)), 0.1)
                lS.C1 = lS.C1:Lerp(CFrame.new(0, 0.5, 0) * CFrame.Angles(math.rad(-10), 0, math.rad(15)), 0.1)
                neck.C1 = neck.C1:Lerp(CFrame.new(0, -0.2, 0) * CFrame.Angles(math.rad(-15), 0, 0), 0.1)
            end
        end
    end
end

        local rS = char:FindFirstChild("RightUpperArm") and char.RightUpperArm:FindFirstChild("RightShoulder")
        local lS = char:FindFirstChild("LeftUpperArm") and char.LeftUpperArm:FindFirstChild("LeftShoulder")

        local rH = char:FindFirstChild("RightUpperLeg") and char.RightUpperLeg:FindFirstChild("RightHip")
        local lH = char:FindFirstChild("LeftUpperLeg") and char.LeftUpperLeg:FindFirstChild("LeftHip")
        local rL = char:FindFirstChild("RightLowerLeg") and char.RightLowerLeg:FindFirstChild("RightKnee")

        if neck then neck.C1 = CFrame.new(0, -0.5, 0) * CFrame.Angles(0, 0, 0) end
        if rH then rH.C1 = CFrame.new(0, -0.5, 0) * CFrame.Angles(0, 0, 0) end
        if lH then lH.C1 = CFrame.new(0, -0.5, 0) * CFrame.Angles(0, 0, 0) end
        if rL then rL.C1 = CFrame.new(0, -0.5, 0) * CFrame.Angles(0, 0, 0) end
        if lL then lL.C1 = CFrame.new(0, -0.5, 0) * CFrame.Angles(0, 0, 0) end
        if torso then torso.C1 = CFrame.new(0, 0, 0) * CFrame.Angles(0, 0, 0) end
    else
        local rS = char:FindFirstChild("Right Shoulder")
        local lS = char:FindFirstChild("Left Shoulder")
        local neck = char:FindFirstChild("Head") and char.Head:FindFirstChild("Neck")
        local rH = char:FindFirstChild("Right Hip")

        if neck then neck.C1 = CFrame.new(0, -0.5, 0) * CFrame.Angles(0, 0, 0) end

        if lH then lH.C1 = CFrame.new(0, -0.5, 0) * CFrame.Angles(0, 0, 0) end

        ToggleFlyBtn.Text = "🛬 DỪNG BAY"

            ColorSequenceKeypoint.new(0, Color3.fromRGB(255, 100, 0)),
            ColorSequenceKeypoint.new(1, Color3.fromRGB(200, 0, 0))

        smoothCameraZoom(flyingFOV, 15)
        blurEffect.Enabled = true
        
        local bg = Instance.new("BodyGyro", root)
        bg.P = 90000
        bg.MaxTorque = Vector3.new(9e9, 9e9, 9e9)
        bg.D = 500
        
        local bv = Instance.new("BodyVelocity", root)
        bv.MaxForce = Vector3.new(9e9, 9e9, 9e9)
        bv.P = 1000
        
        hum.PlatformStand = true
        
        local lastVelocity = Vector3.zero
        
        task.spawn(function()
            while flying and char and char.Parent do
                RunService.RenderStepped:Wait()
                local speed = tonumber(SpeedInput.Text) or 200
                
                if hum.MoveDirection.Magnitude > 0 then
                    local flyDir = (hum.MoveDirection + Vector3.new(0, camera.CFrame.LookVector.Y * 1.5, 0)).Unit
                    local targetVel = flyDir * speed
                    bv.Velocity = bv.Velocity:Lerp(targetVel, 0.1)
                    
                    if bv.Velocity.Magnitude > 1 then
                        local lookDir = bv.Velocity.Unit
                        bg.CFrame = CFrame.new(root.Position, root.Position + lookDir) * CFrame.Angles(math.rad(-70), 0, 0)
                    end
                    
                    setFlyPose(char, bv.Velocity.Magnitude)
                    
                    if flyParticles then
                        flyParticles.Rate = 100
                        flyParticles.Speed = NumberRange.new(5, 10)
                    end
                else
                    bv.Velocity = bv.Velocity:Lerp(Vector3.zero, 0.08)
                    
                    local lookDir = camera.CFrame.LookVector * Vector3.new(1, 0, 1)
                    if lookDir.Magnitude > 0 then
                        bg.CFrame = bg.CFrame:Lerp(CFrame.new(root.Position, root.Position + lookDir), 0.08)
                    end
                    
                    setFlyPose(char, 0)
                    
                    if flyParticles then
                        flyParticles.Rate = 40
                        flyParticles.Speed = NumberRange.new(2, 4)
                    end
                end
                
                lastVelocity = bv.Velocity
            end
            
            if bg then bg:Destroy() end
            if bv then bv:Destroy() end
            hum.PlatformStand = false
            
            resetPose(char)
            
            if flyParticles then 
                flyParticles.Enabled = false
                flyParticles:Destroy()
                flyParticles = nil
            end
            
            if speedLines then
                speedLines:Destroy()
                speedLines = nil
            end
        end)
    else
        ToggleFlyBtn.Text = "🚀 DÙNG BAY"
        flyGradient.Color = ColorSequence.new{
            ColorSequenceKeypoint.new(0, Color3.fromRGB(0, 200, 255)),
            ColorSequenceKeypoint.new(1, Color3.fromRGB(0, 120, 255))
        }
        
        createLandingEffect(char)
        
        if originalCameraSubject then
            camera.CameraSubject = originalCameraSubject
        end
        if originalCameraType then
            camera.CameraType = originalCameraType
        end
        
        smoothCameraZoom(defaultFOV, 0)
        blurEffect.Enabled = false
        
        if flyParticles then 
            flyParticles.Enabled = false
        end
        
        if speedLines then
            speedLines:Destroy()
            speedLines = nil
        end
    end
end)
