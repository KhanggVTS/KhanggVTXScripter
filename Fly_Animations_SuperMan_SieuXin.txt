local function smoothCameraZoom(targetFOV, speed)
    if zoomTween then
        zoomTween:Cancel()
    end
    
    local currentFOV = camera.FieldOfView
    local diff = targetFOV - currentFOV
    
    zoomTween = TweenService:Create(camera, 
        TweenInfo.new(0.5, Enum.EasingStyle.Quad, Enum.EasingDirection.Out),
        {FieldOfView = targetFOV}
    )
    zoomTween:Play()
    
    zoomTween.Completed:Connect(function()
        zoomTween = nil
    end)
end

local function smoothCameraOffset(targetOffset, speed)
    local currentOffset = cameraOffset
    local diff = targetOffset - currentOffset
    
    local startTime = tick()
    local duration = 0.4
    
    local connection
    connection = RunService.RenderStepped:Connect(function()
        local elapsed = tick() - startTime
        local alpha = math.min(elapsed / duration, 1)
        
        local easedAlpha = 1 - (1 - alpha) * (1 - alpha)
        cameraOffset = currentOffset:Lerp(targetOffset, easedAlpha)
        
        if alpha >= 1 then
            connection:Disconnect()
        end
    end)
end

local function setR15Pose(char, mode, alpha)
    local hum = char:FindFirstChildOfClass("Humanoid")
    if not hum then return end
    
    if hum.RigType == Enum.HumanoidRigType.R15 then
        local rS = char:FindFirstChild("RightUpperArm") and char.RightUpperArm:FindFirstChild("RightShoulder")
        local lS = char:FindFirstChild("LeftUpperArm") and char.LeftUpperArm:FindFirstChild("LeftShoulder")
        local neck = char:FindFirstChild("Head") and char.Head:FindFirstChild("Neck")
        local rH = char:FindFirstChild("RightUpperLeg") and char.RightUpperLeg:FindFirstChild("RightHip")
        local lH = char:FindFirstChild("LeftUpperLeg") and char.LeftUpperLeg:FindFirstChild("LeftHip")
        local torso = char:FindFirstChild("UpperTorso") and char.UpperTorso:FindFirstChild("Waist")
        
        if rS and lS and neck then
            local tRS, tLS, tN, tRH, tLH, tTorso
            
            if mode == "Flying" then
                tRS = CFrame.new(0.2, 0.8, -0.2) * CFrame.Angles(math.rad(-150), math.rad(10), math.rad(-30))
                tLS = CFrame.new(-0.2, 0.8, -0.2) * CFrame.Angles(math.rad(-150), math.rad(-10), math.rad(30))
                tN = CFrame.new(0, -0.4, 0.1) * CFrame.Angles(math.rad(-40), 0, 0)
                if rH then tRH = CFrame.new(0, -0.4, 0.1) * CFrame.Angles(math.rad(-20), 0, math.rad(-5)) end
                if lH then tLH = CFrame.new(0, -0.4, 0.1) * CFrame.Angles(math.rad(-20), 0, math.rad(5)) end
                if torso then tTorso = CFrame.new(0, 0, 0) * CFrame.Angles(math.rad(-30), 0, 0) end
            elseif mode == "Hover" then
                tRS = CFrame.new(0, 0.6, 0) * CFrame.Angles(math.rad(-10), 0, math.rad(-15))
                tLS = CFrame.new(0, 0.6, 0) * CFrame.Angles(math.rad(-10), 0, math.rad(15))
                tN = CFrame.new(0, -0.3, 0) * CFrame.Angles(math.rad(-15), 0, 0)
                if rH then tRH = CFrame.new(0, -0.3, 0) * CFrame.Angles(math.rad(-10), 0, 0) end
                if lH then tLH = CFrame.new(0, -0.3, 0) * CFrame.Angles(math.rad(-10), 0, 0) end
            else
                tRS = CFrame.new(0, 0.5, 0) * CFrame.Angles(0, 0, math.rad(-25))
                tLS = CFrame.new(0, 0.5, 0) * CFrame.Angles(0, 0, math.rad(25))
                tN = CFrame.new(0, -0.5, 0) * CFrame.Angles(0, 0, 0)
                if rH then tRH = CFrame.new(0, -0.5, 0) * CFrame.Angles(0, 0, 0) end
                if lH then tLH = CFrame.new(0, -0.5, 0) * CFrame.Angles(0, 0, 0) end
                if torso then tTorso = CFrame.new(0, 0, 0) * CFrame.Angles(0, 0, 0) end
            end
            
            rS.C1 = rS.C1:Lerp(tRS, alpha)
            lS.C1 = lS.C1:Lerp(tLS, alpha)
            neck.C1 = neck.C1:Lerp(tN, alpha)
            if rH and tRH then rH.C1 = rH.C1:Lerp(tRH, alpha) end
            if lH and tLH then lH.C1 = lH.C1:Lerp(tLH, alpha) end
            if torso and tTorso then torso.C1 = torso.C1:Lerp(tTorso, alpha) end
        end
    else
        local rS = char:FindFirstChild("Right Shoulder")
        local lS = char:FindFirstChild("Left Shoulder")
        local neck = char:FindFirstChild("Head") and char.Head:FindFirstChild("Neck")
        
        if rS and lS and neck then
            local tRS, tLS, tN
            
            if mode == "Flying" then
                tRS = CFrame.new(0, 0.5, 0) * CFrame.Angles(math.rad(-150), 0, math.rad(-20))
                tLS = CFrame.new(0, 0.5, 0) * CFrame.Angles(math.rad(-150), 0, math.rad(20))
                tN = CFrame.new(0, -0.3, 0) * CFrame.Angles(math.rad(-40), 0, 0)
            elseif mode == "Hover" then
                tRS = CFrame.new(0, 0.5, 0) * CFrame.Angles(math.rad(-10), 0, math.rad(-15))
                tLS = CFrame.new(0, 0.5, 0) * CFrame.Angles(math.rad(-10), 0, math.rad(15))
                tN = CFrame.new(0, -0.4, 0) * CFrame.Angles(math.rad(-15), 0, 0)
            else
                tRS = CFrame.new(0, 0.5, 0) * CFrame.Angles(0, 0, math.rad(-25))
                tLS = CFrame.new(0, 0.5, 0) * CFrame.Angles(0, 0, math.rad(25))
                tN = CFrame.new(0, -0.5, 0) * CFrame.Angles(0, 0, 0)
            end
            
            rS.C1 = rS.C1:Lerp(tRS, alpha)
            lS.C1 = lS.C1:Lerp(tLS, alpha)
            neck.C1 = neck.C1:Lerp(tN, alpha)
        end
    end
end

ToggleFlyBtn.MouseButton1Click:Connect(function()
    flying = not flying
    local char = player.Character
    if not char then return end
    local root = char:FindFirstChild("HumanoidRootPart")
    local hum = char:FindFirstChildOfClass("Humanoid")
    if not root or not hum then return end
    
    if flying then
        ToggleFlyBtn.Text = "🛬 DỪNG BAY"
        flyGradient.Color = ColorSequence.new{
            ColorSequenceKeypoint.new(0, Color3.fromRGB(255, 100, 0)),
            ColorSequenceKeypoint.new(1, Color3.fromRGB(200, 0, 0))
        }
        
        createTakeoffEffect(char)
        createFlyTrail(char)
        if flyParticles then flyParticles.Enabled = true end
        
        smoothCameraZoom(flyingFOV, 0.5)
        smoothCameraOffset(Vector3.new(0, 0, -3), 0.4)
        
        local bg = Instance.new("BodyGyro", root)
        bg.P = 9e4
        bg.MaxTorque = Vector3.new(9e9, 9e9, 9e9)
        bg.D = 500
        
        local bv = Instance.new("BodyVelocity", root)
        bv.MaxForce = Vector3.new(9e9, 9e9, 9e9)
        bv.P = 1000
        
        hum.PlatformStand = true
        
        local poseAlpha = 0
        
        task.spawn(function()
            while flying and char.Parent do
                RunService.RenderStepped:Wait()
                local speed = tonumber(SpeedInput.Text) or 200
                
                poseAlpha = math.min(poseAlpha + 0.05, 1)
                
                if hum.MoveDirection.Magnitude > 0 then
                    local flyDir = (hum.MoveDirection + Vector3.new(0, camera.CFrame.LookVector.Y * 1.5, 0)).Unit
                    bv.Velocity = bv.Velocity:Lerp(flyDir * speed, 0.12)
                    
                    local lookDir = bv.Velocity.Unit
                    if lookDir.Magnitude > 0 then
                        bg.CFrame = CFrame.new(root.Position, root.Position + lookDir) * CFrame.Angles(math.rad(-70), 0, 0)
                    end
                    
                    setR15Pose(char, "Flying", poseAlpha)
                    
                    if flyParticles then
                        flyParticles.Speed = NumberRange.new(4, 8)
                        flyParticles.Rate = 100
                    end
                else
                    bv.Velocity = bv.Velocity:Lerp(Vector3.zero, 0.1)
                    
                    local lookDir = camera.CFrame.LookVector * Vector3.new(1, 0, 1)
                    if lookDir.Magnitude > 0 then
                        bg.CFrame = bg.CFrame:Lerp(CFrame.new(root.Position, root.Position + lookDir), 0.1)
                    end
                    
                    setR15Pose(char, "Hover", poseAlpha)
                    
                    if flyParticles then
                        flyParticles.Speed = NumberRange.new(2, 4)
                        flyParticles.Rate = 50
                    end
                end
            end
            
            bg:Destroy()
            bv:Destroy()
            hum.PlatformStand = false
            
            local resetPoseAlpha = 0.3
            local resetConnection
            resetConnection = RunService.RenderStepped:Connect(function()
                resetPoseAlpha = resetPoseAlpha - 0.02
                if resetPoseAlpha <= 0 then
                    setR15Pose(char, "Normal", 0)
                    resetConnection:Disconnect()
                else
                    setR15Pose(char, "Hover", resetPoseAlpha)
                end
            end)
            
            if flyParticles then 
                flyParticles.Enabled = false
                flyParticles:Destroy()
                flyParticles = nil
            end
        end)
    else
        ToggleFlyBtn.Text = "🚀 KÍCH HOẠT BAY"
        flyGradient.Color = ColorSequence.new{
            ColorSequenceKeypoint.new(0, Color3.fromRGB(0, 200, 255)),
            ColorSequenceKeypoint.new(1, Color3.fromRGB(0, 120, 255))
        }
        
        createLandingEffect(char)
        
        smoothCameraZoom(defaultFOV, 0.5)
        smoothCameraOffset(defaultCameraOffset, 0.4)
        
        if flyParticles then 
            flyParticles.Enabled = false
        end
    end
end)
