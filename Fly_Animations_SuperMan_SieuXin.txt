local function smoothCameraZoom(targetFOV)
    if zoomTween then zoomTween:Cancel() end
    zoomTween = TweenService:Create(camera, 
        TweenInfo.new(0.6, Enum.EasingStyle.Quad, Enum.EasingDirection.Out),
        {FieldOfView = targetFOV}
    )
    zoomTween:Play()
end

local function createBodyAura(char)
    if bodyAura then bodyAura:Destroy() end
    
    local root = char:FindFirstChild("HumanoidRootPart")
    if not root then return end
    
    bodyAura = Instance.new("Part")
    bodyAura.Size = Vector3.new(4, 5, 2)
    bodyAura.Transparency = 0.7
    bodyAura.CanCollide = false
    bodyAura.Anchored = false
    bodyAura.Material = Enum.Material.Neon
    bodyAura.Color = Color3.fromRGB(0, 200, 255)
    bodyAura.Parent = char
    
    local weld = Instance.new("WeldConstraint", bodyAura)
    weld.Part0 = bodyAura
    weld.Part1 = root
    
    local auraLight = Instance.new("PointLight", bodyAura)
    auraLight.Color = Color3.fromRGB(0, 200, 255)
    auraLight.Range = 10
    auraLight.Brightness = 2
    
    local auraParticles = Instance.new("ParticleEmitter", bodyAura)
    auraParticles.Texture = "rbxassetid://77302218896291"
    auraParticles.Rate = 30
    auraParticles.Lifetime = NumberRange.new(0.5, 1)
    auraParticles.Speed = NumberRange.new(1, 2)
    auraParticles.SpreadAngle = Vector2.new(360, 360)
    auraParticles.Color = ColorSequence.new{
        ColorSequenceKeypoint.new(0, Color3.fromRGB(0, 200, 255)),
        ColorSequenceKeypoint.new(0.5, Color3.fromRGB(150, 0, 255)),
        ColorSequenceKeypoint.new(1, Color3.fromRGB(255, 0, 150))
    }
    auraParticles.Size = NumberSequence.new{
        NumberSequenceKeypoint.new(0, 0.3),
        NumberSequenceKeypoint.new(1, 0)
    }
    auraParticles.Transparency = NumberSequence.new{
        NumberSequenceKeypoint.new(0, 0.5),
        NumberSequenceKeypoint.new(1, 1)
    }
    auraParticles.LightEmission = 1
    
    task.spawn(function()
        while bodyAura and bodyAura.Parent and flying do
            local hue = (tick() * 0.5) % 1
            local color = Color3.fromHSV(hue, 0.9, 1)
            bodyAura.Color = color
            auraLight.Color = color
            auraParticles.Color = ColorSequence.new{
                ColorSequenceKeypoint.new(0, color),
                ColorSequenceKeypoint.new(0.5, Color3.fromHSV((hue + 0.3) % 1, 0.9, 1)),
                ColorSequenceKeypoint.new(1, Color3.fromHSV((hue + 0.6) % 1, 0.9, 1))
            }
            task.wait(0.1)
        end
    end)
end

local function resetAllJoints(char)
    local hum = char:FindFirstChildOfClass("Humanoid")
    if not hum then return end
    
    if hum.RigType == Enum.HumanoidRigType.R15 then
        local joints = {
            char:FindFirstChild("RightUpperArm") and char.RightUpperArm:FindFirstChild("RightShoulder"),
            char:FindFirstChild("LeftUpperArm") and char.LeftUpperArm:FindFirstChild("LeftShoulder"),
            char:FindFirstChild("Head") and char.Head:FindFirstChild("Neck"),
            char:FindFirstChild("RightUpperLeg") and char.RightUpperLeg:FindFirstChild("RightHip"),
            char:FindFirstChild("LeftUpperLeg") and char.LeftUpperLeg:FindFirstChild("LeftHip"),
            char:FindFirstChild("RightLowerLeg") and char.RightLowerLeg:FindFirstChild("RightKnee"),
            char:FindFirstChild("LeftLowerLeg") and char.LeftLowerLeg:FindFirstChild("LeftKnee"),
            char:FindFirstChild("UpperTorso") and char.UpperTorso:FindFirstChild("Waist")
        }
        
        for _, joint in ipairs(joints) do
            if joint then
                if joint.Name == "RightShoulder" then
                    joint.C1 = CFrame.new(0, 0.5, 0) * CFrame.Angles(0, 0, math.rad(-25))
                elseif joint.Name == "LeftShoulder" then
                    joint.C1 = CFrame.new(0, 0.5, 0) * CFrame.Angles(0, 0, math.rad(25))
                elseif joint.Name == "Neck" then
                    joint.C1 = CFrame.new(0, -0.5, 0)
                else
                    joint.C1 = CFrame.new(0, 0, 0)
                end
            end
        end
    else
        local rS = char:FindFirstChild("Right Shoulder")
        local lS = char:FindFirstChild("Left Shoulder")
        local neck = char:FindFirstChild("Head") and char.Head:FindFirstChild("Neck")
        local rH = char:FindFirstChild("Right Hip")
        local lH = char:FindFirstChild("Left Hip")
        
        if rS then rS.C1 = CFrame.new(0, 0.5, 0) end
        if lS then lS.C1 = CFrame.new(0, 0.5, 0) end
        if neck then neck.C1 = CFrame.new(0, -0.5, 0) end
        if rH then rH.C1 = CFrame.new(0, -0.5, 0) end
        if lH then lH.C1 = CFrame.new(0, -0.5, 0) end
    end
end

ToggleFlyBtn.MouseButton1Click:Connect(function()
    flying = not flying
    local char = player.Character
    if not char then return end
    local root = char:FindFirstChild("HumanoidRootPart")
    local hum = char:FindFirstChildOfClass("Humanoid")
    if not root or not hum then return end
    
    if flying then
        ToggleFlyBtn.Text = "🛬 DỪNG BAY"
        flyGradient.Color = ColorSequence.new{
            ColorSequenceKeypoint.new(0, Color3.fromRGB(255, 100, 0)),
            ColorSequenceKeypoint.new(1, Color3.fromRGB(200, 0, 0))
        }
        
        createTakeoffEffect(char)
        createFlyTrail(char)
        createBodyAura(char)
        if flyParticles then flyParticles.Enabled = true end
        
        originalCameraSubject = camera.CameraSubject
        originalCameraType = camera.CameraType
        camera.CameraSubject = root
        camera.CameraType = Enum.CameraType.Custom
        
        smoothCameraZoom(flyingFOV)
        
        local bg = Instance.new("BodyGyro", root)
        bg.P = 90000
        bg.MaxTorque = Vector3.new(9e9, 9e9, 9e9)
        bg.D = 500
        
        local bv = Instance.new("BodyVelocity", root)
        bv.MaxForce = Vector3.new(9e9, 9e9, 9e9)
        bv.P = 1000
        
        hum.PlatformStand = true
        
        local legTime = 0
        
        legSwingConnection = RunService.RenderStepped:Connect(function(deltaTime)
            if not flying or not char or not char.Parent then return end
            
            legTime = legTime + deltaTime * 10
            
            local speed = tonumber(SpeedInput.Text) or 200
            
            if hum.MoveDirection.Magnitude > 0 then
                local flyDir = (hum.MoveDirection + Vector3.new(0, camera.CFrame.LookVector.Y * 1.8, 0)).Unit
                bv.Velocity = bv.Velocity:Lerp(flyDir * speed, 0.12)
                
                if bv.Velocity.Magnitude > 1 then
                    local lookDir = bv.Velocity.Unit
                    bg.CFrame = CFrame.new(root.Position, root.Position + lookDir) * CFrame.Angles(math.rad(-65), 0, 0)
                end
                
                if flyParticles then
                    flyParticles.Rate = 120
                    flyParticles.Speed = NumberRange.new(6, 12)
                end
            else
                bv.Velocity = bv.Velocity:Lerp(Vector3.zero, 0.1)
                
                local lookDir = camera.CFrame.LookVector * Vector3.new(1, 0, 1)
                if lookDir.Magnitude > 0 then
                    bg.CFrame = bg.CFrame:Lerp(CFrame.new(root.Position, root.Position + lookDir), 0.1)
                end
                
                if flyParticles then
                    flyParticles.Rate = 50
                    flyParticles.Speed = NumberRange.new(3, 5)
                end
            end
            
            if hum.RigType == Enum.HumanoidRigType.R15 then
                local rH = char:FindFirstChild("RightUpperLeg") and char.RightUpperLeg:FindFirstChild("RightHip")
                local lH = char:FindFirstChild("LeftUpperLeg") and char.LeftUpperLeg:FindFirstChild("LeftHip")
                local rL = char:FindFirstChild("RightLowerLeg") and char.RightLowerLeg:FindFirstChild("RightKnee")
                local lL = char:FindFirstChild("LeftLowerLeg") and char.LeftLowerLeg:FindFirstChild("LeftKnee")
                local rS = char:FindFirstChild("RightUpperArm") and char.RightUpperArm:FindFirstChild("RightShoulder")
                local lS = char:FindFirstChild("LeftUpperArm") and char.LeftUpperArm:FindFirstChild("LeftShoulder")
                local neck = char:FindFirstChild("Head") and char.Head:FindFirstChild("Neck")
                local torso = char:FindFirstChild("UpperTorso") and char.UpperTorso:FindFirstChild("Waist")
                
                if bv.Velocity.Magnitude > 5 then
                    local legSwing = math.sin(legTime) * 35
                    local armSwing = math.sin(legTime) * 8
                    
                    if rH then rH.C1 = CFrame.new(0, -0.5, 0.1) * CFrame.Angles(math.rad(-20 + legSwing), 0, math.rad(-3)) end
                    if lH then lH.C1 = CFrame.new(0, -0.5, 0.1) * CFrame.Angles(math.rad(-20 - legSwing), 0, math.rad(3)) end
                    if rL then rL.C1 = CFrame.new(0, -0.5, 0) * CFrame.Angles(math.rad(math.abs(legSwing) * 0.4), 0, 0) end
                    if lL then lL.C1 = CFrame.new(0, -0.5, 0) * CFrame.Angles(math.rad(math.abs(legSwing) * 0.4), 0, 0) end
                    
                    if rS then rS.C1 = CFrame.new(0.2, 0.7, -0.3) * CFrame.Angles(math.rad(-160 + armSwing), math.rad(15), math.rad(-30)) end
                    if lS then lS.C1 = CFrame.new(-0.2, 0.7, -0.3) * CFrame.Angles(math.rad(-160 - armSwing), math.rad(-15), math.rad(30)) end
                    if neck then neck.C1 = CFrame.new(0, -0.4, 0.2) * CFrame.Angles(math.rad(-35), 0, 0) end
                    if torso then torso.C1 = CFrame.new(0, 0, 0) * CFrame.Angles(math.rad(-25), 0, 0) end
                else
                    if rH then rH.C1 = rH.C1:Lerp(CFrame.new(0, -0.4, 0) * CFrame.Angles(math.rad(-10), 0, 0), 0.1) end
                    if lH then lH.C1 = lH.C1:Lerp(CFrame.new(0, -0.4, 0) * CFrame.Angles(math.rad(-10), 0, 0), 0.1) end
                    if rL then rL.C1 = rL.C1:Lerp(CFrame.new(0, -0.5, 0) * CFrame.Angles(0, 0, 0), 0.1) end
                    if lL then lL.C1 = lL.C1:Lerp(CFrame.new(0, -0.5, 0) * CFrame.Angles(0, 0, 0), 0.1) end
                    if rS then rS.C1 = rS.C1:Lerp(CFrame.new(0, 0.6, 0) * CFrame.Angles(math.rad(-15), 0, math.rad(-20)), 0.1) end
                    if lS then lS.C1 = lS.C1:Lerp(CFrame.new(0, 0.6, 0) * CFrame.Angles(math.rad(-15), 0, math.rad(20)), 0.1) end
                    if neck then neck.C1 = neck.C1:Lerp(CFrame.new(0, -0.3, 0) * CFrame.Angles(math.rad(-20), 0, 0), 0.1) end
                    if torso then torso.C1 = torso.C1:Lerp(CFrame.new(0, 0, 0) * CFrame.Angles(math.rad(-10), 0, 0), 0.1) end
                end
            else
                local rH = char:FindFirstChild("Right Hip")
                local lH = char:FindFirstChild("Left Hip")
                local rS = char:FindFirstChild("Right Shoulder")
                local lS = char:FindFirstChild("Left Shoulder")
                local neck = char:FindFirstChild("Head") and char.Head:FindFirstChild("Neck")
                
                if bv.Velocity.Magnitude > 5 then
                    local legSwing = math.sin(legTime) * 30
                    
                    if rH then rH.C1 = CFrame.new(0, -0.5, 0) * CFrame.Angles(math.rad(-15 + legSwing), 0, 0) end
                    if lH then lH.C1 = CFrame.new(0, -0.5, 0) * CFrame.Angles(math.rad(-15 - legSwing), 0, 0) end
                    if rS then rS.C1 = CFrame.new(0, 0.5, 0) * CFrame.Angles(math.rad(-160), 0, math.rad(-25)) end
                    if lS then lS.C1 = CFrame.new(0, 0.5, 0) * CFrame.Angles(math.rad(-160), 0, math.rad(25)) end
                    if neck then neck.C1 = CFrame.new(0, -0.4, 0) * CFrame.Angles(math.rad(-35), 0, 0) end
                end
            end
        end)
        
        task.spawn(function()
            while flying and char and char.Parent do
                RunService.RenderStepped:Wait()
            end
            
            if bg then bg:Destroy() end
            if bv then bv:Destroy() end
            if legSwingConnection then legSwingConnection:Disconnect() end
            hum.PlatformStand = false
            
            resetAllJoints(char)
            
            if flyParticles then 
                flyParticles.Enabled = false
                flyParticles:Destroy()
                flyParticles = nil
            end
            
            if bodyAura then
                bodyAura:Destroy()
                bodyAura = nil
            end
        end)
    else
        ToggleFlyBtn.Text = "🚀 DÙNG BAY"
